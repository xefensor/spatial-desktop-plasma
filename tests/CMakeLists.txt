cmake_minimum_required(VERSION 3.22)
project(SpatialAppsChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Qml Quick Test)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
add_executable(test-apps test_apps.cpp)
target_link_libraries(test-apps PRIVATE Qt6::Widgets Qt6::Quick Qt6::Test Plasma::Plasma Plasma::PlasmaQuick)
target_compile_definitions(test-apps PRIVATE SPATIAL_PLASMA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/..")
enable_testing()
add_test(NAME apps-qml COMMAND test-apps)
set_tests_properties(apps-qml PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QUICK_CONTROLS_STYLE=Fusion")
